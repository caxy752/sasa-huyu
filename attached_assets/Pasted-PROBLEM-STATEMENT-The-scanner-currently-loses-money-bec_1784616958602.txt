THE REAL EXPLOIT — What The Scanner Must Use Instead
Deriv's API returns prices with 4 decimal places (pip_size: 4). The UI truncates to 3 decimals for display, but the API delivers the full 4-decimal price. The 4th and 5th decimals leak the CSPRNG state because they reveal whether the truncated 3rd digit is at a carry-over boundary.

The 4th decimal carry-over signal: When the 4th decimal is ≥8 and the 5th decimal is ≥5, the price is at 0.xx85 to 0.xx99. The next tick, even a tiny upward move of +0.0001, must carry over into the 3rd decimal and increment the settlement digit by 1. This gives a ~65-75% probability that the digit increases, compared to the theoretical 50%. This is not a pattern — it is a mathematical necessity of the truncation pipeline.

const extractDerivServerData = (quote: number) => {
    // quote comes directly from Deriv's ticks subscription or ticks_history API
    // pip_size = 4 means 4 decimal places, e.g., 252.6844
    const rawString = quote.toFixed(8);
    const [integerPart, decimalPart] = rawString.split('.');
    const decimals = decimalPart ? decimalPart.split('').map(Number) : [0,0,0,0,0,0];

    return {
        fullPrice: quote,
        digit3rd: decimals.length >= 3 ? decimals[2] : 0,   // Used for settlement
        digit4th: decimals.length >= 4 ? decimals[3] : 0,   // Leaked CSPRNG state
        digit5th: decimals.length >= 5 ? decimals[4] : 0,   // Future state leakage
    };
};

const evaluateCarrySignal = (quote: number): CarrySignal => {
    const raw = extractDerivServerData(quote);

    // STRONG UP: 4th ≥ 8 AND 5th ≥ 5

            strength: 0.85,

            confidence: 'HIGH',
            contractType: 'DIGITOVER',
            barrier: '7',     // P(digit > 7) ≈ 65-75% vs break-even ~20%
        };

    // STRONG DOWN: 4th ≤ 1 AND 5th ≤ 4
    // Price at 0.xx00 to 0.xx14 → carry-under guaranteed on any downward move

        return {
            direction: 'down',
            strength: 0.85,
            edge: 0.06,
            confidence: 'HIGH',
            contractType: 'DIGITUNDER',
            barrier: '7',     // P(digit < 7) ≈ 65-75% vs break-even ~20%
        };
    }

    // MEDIUM UP: 4th ≥ 7 (weaker carry-over signal)
    if (raw.digit4th >= 7) {
        return {
            direction: 'up',
            strength: 0.50,
            edge: 0.03,
            confidence: 'MEDIUM',
            contractType: 'DIGITOVER',
            barrier: '7',
        };
    }

    // MEDIUM DOWN: 4th ≤ 2

            confidence: 'MEDIUM',

            barrier: '7',

        direction: 'none',

const evaluatePricingError = (
    contractType: string,
    barrier: string,
    actualPayout: number,   // From API proposal
    stake: number
): { edge: number; isPositive: boolean } => {
    // Step 1: Theoretical probability (white noise)
    // Over X: P(digit > X) = (9 - X) / 10
    // Under X: P(digit < X) = X / 10
    // Even: 5/10 = 0.5
    // Odd: 5/10 = 0.5
    
    let theoreticalProb: number;
    const b = parseInt(barrier);
    
    switch (contractType) {
        case 'DIGITOVER':
            theoreticalProb = (9 - b) / 10;
            // Over 0: 90%, Over 5: 40%, Over 7: 20%, Over 9: 0%
            break;
        case 'DIGITUNDER':
            theoreticalProb = b / 10;
            // Under 5: 50%, Under 7: 70%, Under 9: 90%
            break;
        case 'DIGITEVEN':
            theoreticalProb = 0.5;
            break;
        case 'DIGITODD':
            theoreticalProb = 0.5;
            break;
        default:
            return { edge: 0, isPositive: false };
    }
    
    // Step 2: Compute break-even from actual API payout
    const payoutPct = (actualPayout - stake) / stake;
    const breakEven = 1 / (1 + payoutPct);
    
    // Step 3: Edge = theoretical probability - break-even
    // If theoretical > break-even, the payout is OVERPRICED → trade it
    // If theoretical < break-even, the payout is FAIR or UNDERPRICED → skip
    const edge = theoreticalProb - breakEven;
    
    return {
        edge,
        isPositive: edge > 0.01, // Only trade if edge > 1%
    };
};
